	\begin{enumerate}[label=1.\arabic*)]
		\item \textbf{Modelagem de Epidemias}. Uma cidade está enfrentando uma epidemia, e a probabilidade de uma pessoa ser infectada é de 5\%. Suponha que a cidade tenha 50.000 habitantes e você é contratado para modelar a propagação da doença. Considere que o número de pessoas infectadas, $X$, segue a distribuição binomial com $n=50\,000$ habitantes e $p=0.05$.
		\begin{enumerate}
			\item Calcule $P(X=2\,600)$, a probabilidade de exatamente 2\,600 pessoas serem infectadas.
			\item Calcule $P(X\leq 2\,600)$, a probabilidade de até 2\,600 pessoas serem infectadas.
			\item Simule no R 1\,000 amostras e faça um histograma da distribuição de pessoas infectadas [use \texttt{set.seed(12345)}].
		\end{enumerate}
		
		\item \textbf{Análise de Fluxo de Veículos em um Pedágio.} Uma concessionária de rodovias está estudando o fluxo de veículos em um pedágio. Em média, passam 10 veículos por minuto. Para otimizar a operação do pedágio, a concessionária precisa entender como esse fluxo varia. Considere que o número de veículos que passam por minuto segue uma distribuição de Poisson com parâmetro $\lambda=10$ (média de 10 veículos por minuto).
		\begin{enumerate}
			\item Calcule $P(X=8)$, a probabilidade de exatamente 8 veículos passarem pelo pedágio em um minuto.
			\item Calcule $P(5<X\leq 15)$, a probabilidade de entre 6 até 15 veículos passarem pelo pedágio em um minuto.
			\item Simule no R 1\,000 amostras de uma distribuição Poisson com  $\lambda=10$ e faça um histograma dos dados [use \texttt{set.seed(12345)}].
		\end{enumerate}
		
		\item \textbf{Análise de Temperatura em uma Sala.} Um sistema de climatização em um escritório mantém a temperatura ambiente controlada entre 18$^\circ$C e 24$^\circ$C. A empresa deseja entender a variação da temperatura ao longo do dia para garantir o conforto dos funcionários. Considere que a temperatura $X$ na sala segue uma distribuição Uniforme(18,24).
		\begin{enumerate}
			\item Calcule $P(X\leq 20)$, a probabilidade de a temperatura ser menor ou igual a 20$^\circ$C.
			\item Calcule $P(19\leq X\leq 22)$, a probabilidade de a temperatura estar entre 19$^\circ$C e 22$^\circ$C.
			\item Simule no R 1\,000 amostras de uma distribuição Uniforme(18,24) e faça um histograma dos dados [use \texttt{set.seed(12345)}].
		\end{enumerate}
		
		\item \textbf{Análise de Risco em Investimentos.} Uma carteira de investimentos tem um retorno anual que segue uma distribuição normal com média de 7\% e desvio padrão de 2\% (variância de 4\%$^2$).
		\begin{enumerate}
			\item Calcule  $P(X<0)$, a probabilidade de a carteira ter um retorno negativo em um ano.
			\item Calcule $P(5\leq X\leq 10)$, a probabilidade de o retorno ser entre 5\% e 10\%.
			\item Calcule $P(X\geq6)$, a probabilidade de um pacote ser entregue em 6 dias ou mais.
			\item Simule no R 1\,000 amostras de retornos anuais e faça um histograma [use \texttt{set.seed(12345)}].
		\end{enumerate}
		
		\item \textbf{Análise do Tempo de Atendimento em um Call Center.} Um \textit{call center} está analisando o tempo de atendimento dos clientes para otimizar seus recursos. O tempo de atendimento $T$ é uma variável aleatória que segue uma distribuição exponencial com uma taxa média de  $\alpha=0.2$ atendimentos por minuto (o que equivale a um tempo médio de atendimento de 5 minutos).
		\begin{enumerate}
			\item Calcule $P(T\leq3)$, a probabilidade de que o tempo de atendimento seja de até 3 minutos.
			\item Calcule $P(T\geq 10)$, a probabilidade de que o tempo de atendimento seja superior a 10 minutos.
			\item Simule no R 1\,000 amostras de uma distribuição exponencial com $\lambda=0.2$ e faça um histograma dos dados [use \texttt{set.seed(12345)}].
		\end{enumerate}
		
	\end{enumerate}
