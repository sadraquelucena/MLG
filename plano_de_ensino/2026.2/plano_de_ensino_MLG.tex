\begin{itemize}[label={},itemindent=-2em,leftmargin=2em]
\item {\bf Ementa:} Família exponencial de distribuições. Ajuste de modelos. Função de ligação. Modelos lineares generalizados especiais. Estimação. A função desvio. Testes de hipóteses. Análise de dados reais. Análise dos resíduos. Técnicas de diagnóstico.

\item {\bf Objetivos:} Fornecer ao discente, o conhecimento sobre Modelos Lineares Generalizados e suas aplicações em diversas áreas.

\item {\bf Metodologia:} Serão ministradas aulas teóricas expositivas; utilizados recursos visuais; resolução de exercícios em sala de aula; utilização de software para estimação de MLGs; solicitação de atividades extraclasse.

\item {\bf Habilidades e Competências:} Dentro das perspectivas o aluno terá as qualificações exigidas para a utilização das técnicas apresentadas no curso de forma a ser habilitado a resolver ou mesmo buscar novos conceitos de modelos para resolução de um problema apresentado. O cronograma de aulas encontra-se no Quadro 1.

\item {\bf Avaliação:} Serão realizadas avaliações ao longo do semestre. O(A) aluno(a) cuja média aritmética das notas obtidas for igual ou superior a 5,0 será considerado(a) aprovado(a). Caso a média aritmética das três avaliações seja inferior a 5,0, o(a) aluno(a) será considerado(a) reprovado(a).

\noindent {\bf Hora-trabalho:} Ao final de cada aula serão indicados exercícios para serem resolvidos, bem como exigidos relatórios de modelagem de dados como forma de fixação do conteúdo exposto em sala de aula.
\end{itemize}

\noindent {\bf Conteúdo:}

\begin{enumerate}
	\item Família Exponencial de Distribuições
	\begin{enumerate}[label*=\arabic*.]
		\item Introdução
		\item Família exponencial uniparamétrica e multiparamétrica
	\end{enumerate}
	
	\item Modelo Linear Generalizado
	\begin{enumerate}[label*=\arabic*.]
		\item Introdução
		\item Exemplos de motivação
		\item Definição
	\end{enumerate}
	
	\item Estimação
	\begin{enumerate}[label*=\arabic*.]
		\item O algoritmo de estimação
		\item Estimação em modelos especiais
		\item Resultados adicionais na estimação
		\item Seleção do modelo
	\end{enumerate}
	
	\item Métodos de Inferência
	\begin{enumerate}[label*=\arabic*.]
		\item Distribuição dos estimadores dos parâmetros
		\item Função desvio e estatística de Pearson generalizada
		\item Análise do desvio e seleção de modelos
		\item Estimação do parâmetro de dispersão
		\item Seleção da Função de Ligação
	\end{enumerate}
	
	\item Resíduos
	\begin{enumerate}[label*=\arabic*.]
		\item Introdução
		\item Técnicas para verificação do ajuste de um modelo
		\item Análise dos resíduos e diagnóstico para o modelo clássico de regressão
		\item Análise dos resíduos e diagnóstico para modelos lineares generalizados
		\item Verificações das funções de ligação e adequação da função de variância
	\end{enumerate}
	
	\item Aplicações a dados contínuos
	\item Aplicações a dados discretos
\end{enumerate}
