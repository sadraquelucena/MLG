\begin{center} {\bf Lista de Exercícios 2} \end{center}
\vspace{0.8cm}

	\begin{enumerate}[label=2.\arabic*)]
		\item Na representação da família exponencial, sabemos que a esperança e a variância de uma variável aleatória $Y$ podem ser obtidas diretamente pelas derivadas da função $b(\theta)$, tais que $E[Y] = b'(\theta)$ e $Var[Y] = \phi \, b''(\theta).$ Com base nas funções $b(\theta)$, no parâmetro canônico $\theta$ e no parâmetro de dispersão $\phi$ encontrados na lista anterior, obtenha $E[Y]$ e $Var[Y]$ para cada uma das distribuições abaixo. Expresse os resultados finais em termos dos parâmetros originais da distribuição.
		
		\begin{enumerate}
			\item $Y\sim\text{Exponencial}(\lambda)$, cuja função de densidade é
			$$
			f(y;\lambda)=\lambda e^{-\lambda y},
			\qquad y>0.
			$$
			
			\item $Y\sim\text{Logarítmica}(\rho)$, cuja função de probabilidade é
			$$
			f(y;\rho)
			=
			\frac{\rho^y}{-y\ln(1-\rho)},
			\qquad y\in\{1,2,3,\ldots\}.
			$$
			
			\item $Y\sim\text{Binomial}(m,\mu)$, cuja função de probabilidade é
			$$
			P(Y=y)
			=
			\binom{m}{y}\mu^y(1-\mu)^{m-y},
			\qquad y=0,1,\ldots,m.
			$$
			
			\item $Y\sim\text{Gama}(\mu,\phi)$, cuja função de densidade é
			$$
			f(y;\mu,\phi)
			=
			\frac{1}{\Gamma(1/\phi)}
			\left(\frac{1}{\phi\mu}\right)^{1/\phi}
			y^{1/\phi-1}
			\exp\left(-\frac{y}{\phi\mu}\right),
			\qquad y>0.
			$$
			
			\item $Y\sim\text{Normal inversa}(\mu,\phi)$, cuja função de densidade é
			$$
			f(y;\mu,\phi)
			=
			\frac{1}{\sqrt{2\pi\phi y^3}}
			\exp\left\{
			-\frac{(y-\mu)^2}{2\phi\mu^2y}
			\right\},
			\qquad y>0.
			$$
		\end{enumerate}
		
		\item 
		
		\begin{enumerate}
			\item A função de variância, denotada por $V(\mu)$, é definida como $b''(\theta)$ expressa em termos da média $\mu$. Ela descreve como a variância da distribuição muda em função da sua média.
			
			\item Para todas as distribuições pertencentes à família exponencial, a variância de $Y$ depende obrigatoriamente do valor da sua média $\mu$.
			
			\item Na distribuição Poisson (caso particular onde $\phi = 1$), a função de variância é $V(\mu) = \mu$, o que indica que a média e a variância são iguais.
			
			\item O parâmetro de dispersão $\phi$ é sempre uma constante conhecida a priori em qualquer modelo linear generalizado.
		\end{enumerate}
		
	\end{enumerate}
	
	\newpage
	
	\noindent Respostas:
	
	\begin{enumerate}[label=2.\arabic*)]
		\item 
		
		\begin{enumerate}
			\item $E[Y] = \frac{1}{\lambda}$, $Var[Y]=\frac{1}{\lambda^2}$
			
			\item $E[Y] = \frac{-\rho}{(1-\rho)\ln(1-\rho)}$, $Var[Y] = \frac{-\rho\ln(1-\rho) - \rho^2}{(1-\rho)^2 [\ln(1-\rho)]^2}$
			
			\item $E[Y] = m\mu$, $Var[Y] = m\mu(1-\mu)$
			
			\item $E[Y] = \mu$, $Var[Y] = \phi\mu^2$
			
			\item $E[Y] = \mu$, $Var[Y] = \phi\mu^3$
		\end{enumerate}
		
		\item Julgue cada afirmação como verdadeira (V) ou falsa (F). Justifique as afirmações falsas.
		
		\begin{enumerate}
			\item \textbf{(V)}
			
			\item \textbf{(F)} \textit{Justificativa:} Nem todas as distribuições têm a variância dependente da média. O exemplo clássico é a distribuição Normal. Na família exponencial para a Normal, $b(\theta) = \frac{\theta^2}{2}$, logo $b''(\theta) = 1$. A função de variância é $V(\mu) = 1$ (uma constante) e a variância final é $Var(Y) = \phi \cdot 1 = \sigma^2$, que independe da média $\mu$.
			
			\item \textbf{(V)}
			
			\item \textbf{(F)} \textit{Justificativa:} Em modelos como Normal, Gama e Normal Inversa, o parâmetro $\phi$ costuma ser um parâmetro incidental (desconhecido) que precisa ser estimado a partir dos dados, geralmente através do desvio ou da estatística qui-quadrado de Pearson. Apenas em distribuições como Binomial e Poisson assumimos estritamente $\phi = 1$ a priori.
		\end{enumerate}
		
	\end{enumerate}
